cmake_minimum_required(VERSION 3.24)
project(KF2VREngineerAssetBridge LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The pinned KF2 editor helper requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(KF2VR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${KF2VR_ROOT}/third_party/minhook" "${CMAKE_BINARY_DIR}/minhook")
add_library(kf2vr_engineer_asset_bridge SHARED EngineerAssetBridge.cpp)
target_compile_definitions(kf2vr_engineer_asset_bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(kf2vr_engineer_asset_bridge PRIVATE minhook bcrypt)
set_target_properties(kf2vr_engineer_asset_bridge PROPERTIES OUTPUT_NAME KF2VREngineerAssetBridge PREFIX "")
