# Frame data / NullBackend remain available without Windows or the SDK.
if(WIN32 AND KF2VR_HAVE_OPENXR)
  add_library(kf2vr_xr STATIC src/OpenXrD3D11Backend.cpp)
  target_include_directories(kf2vr_xr PUBLIC include)
  target_compile_features(kf2vr_xr PUBLIC cxx_std_20)
  target_compile_definitions(kf2vr_xr PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
  target_compile_definitions(kf2vr_xr PUBLIC KF2VR_HAS_OPENXR_D3D11=1)
  target_link_libraries(kf2vr_xr PUBLIC kf2vr_core PRIVATE openxr_loader d3d11 dxgi)
else()
  add_library(kf2vr_xr INTERFACE)
  target_include_directories(kf2vr_xr INTERFACE include)
  target_link_libraries(kf2vr_xr INTERFACE kf2vr_core)
  target_compile_definitions(kf2vr_xr INTERFACE KF2VR_HAS_OPENXR_D3D11=0)
endif()
